    62→const FOLDER_META: Record<
    63→  Folder | "STARRED" | "IMPORTANT" | "SNOOZED" | "NOW" | "REPLY" | "WAITING" | "RECEIPTS" | "SUBSCRIPTIONS" | "COMMITMENTS" | "PEOPLE" | "ANALYTICS" | "COMMAND_CENTER",
    64→  { label: string; icon: React.ComponentType<{ className?: string }> }

    86→
    87→export function EmailList({ onOpenEmail }: { onOpenEmail: (id: string) => void }) {
    88→  const folder = useMailStore((s) => s.folder);
    89→  const selectedLabel = useMailStore((s) => s.selectedLabel);
    90→  const searchQuery = useMailStore((s) => s.searchQuery);
    91→  const selectedEmailId = useMailStore((s) => s.selectedEmailId);
    92→  const setSelectedEmailId = useMailStore((s) => s.setSelectedEmailId);
    93→  const openEditDraft = useMailStore((s) => s.openEditDraft);
    94→  const inboxTab = useMailStore((s) => s.inboxTab);
    95→  const setInboxTab = useMailStore((s) => s.setInboxTab);
    96→  const inboxTabsSetting = useSettings((s) => s.inboxTabs);
    97→  const { data, isLoading, isError, refetch, isFetching } = useEmailList();
    98→  const invalidate = useInvalidateMail();
    99→  const [selected, setSelected] = useState<Set<string>>(new Set());
   100→  const [emptyConfirmOpen, setEmptyConfirmOpen] = useState(false);
   101→
   102→  async function handleEmptyFolder() {
   103→    try {
   104→      await Promise.all(
   105→        emails.map((e) =>
   106→          fetch(`/api/emails/${e.id}`, { method: "DELETE" })
   107→        )
   108→      );
   109→      setSelected(new Set());
   110→      setEmptyConfirmOpen(false);
   111→      invalidate();
   112→      toast({
   113→        title: `${folder === "TRASH" ? "Trash" : "Spam"} emptied`,
   114→        duration: 1500,
   115→      });
   116→    } catch {
   117→      toast({ title: "Could not empty folder", variant: "destructive" });
   118→    }
   119→  }
   120→
   121→  const allEmails: Email[] = useMemo(() => data?.emails ?? [], [data]);
   122→  const meta = FOLDER_META[folder];
   123→
   124→  // Inbox category tabs: filter the list by the selected tab when enabled
   125→  const showTabs =
   126→    folder === "INBOX" &&
   127→    !searchQuery &&
   128→    !selectedLabel &&
   129→    inboxTabsSetting === "categories";
   130→  const emails: Email[] = useMemo(() => {
   131→    if (!showTabs || inboxTab === "ALL") return allEmails;
   132→    return allEmails.filter((e) => deriveCategory(e) === inboxTab);
   133→  }, [allEmails, showTabs, inboxTab]);
   134→
   135→  const tabCounts: Record<Category | "ALL", number> = useMemo(() => {
   136→    const c: Record<Category | "ALL", number> = {
   137→      ALL: allEmails.length,
   138→      PRIMARY: 0,
   139→      PROMOTIONS: 0,
   140→      SOCIAL: 0,
   141→      UPDATES: 0,
   142→    };
   143→    for (const e of allEmails) {
   144→      c[deriveCategory(e)] += 1;
   145→    }
   146→    return c;
   147→  }, [allEmails]);
   148→
   149→  const allChecked = emails.length > 0 && selected.size === emails.length;
   150→  const someChecked = selected.size > 0 && selected.size < emails.length;
   151→
   152→  function toggleSelectAll() {
   153→    if (allChecked) setSelected(new Set());
   154→    else setSelected(new Set(emails.map((e) => e.id)));
   155→  }
   156→
   157→  function toggleSelect(id: string) {
   158→    setSelected((prev) => {
   159→      const next = new Set(prev);
   160→      if (next.has(id)) next.delete(id);
   161→      else next.add(id);
   162→      return next;
   163→    });
   164→  }
   165→
   166→  async function bulkUpdate(payload: Record<string, unknown>, msg: string) {
   167→    if (selected.size === 0) return;
   168→    try {
   169→      const res = await fetch("/api/emails", {
   170→        method: "PATCH",
   171→        headers: { "Content-Type": "application/json" },
   172→        body: JSON.stringify({ ids: Array.from(selected), ...payload }),
   173→      });
   174→      if (!res.ok) throw new Error("Update failed");
   175→      setSelected(new Set());
   176→      invalidate();
   177→      toast({ title: msg });
   178→    } catch {
   179→      toast({ title: "Action failed", variant: "destructive" });
   180→    }
   181→  }
   182→
   183→  // Capture prev state per selected email, apply a bulk change, show an undo
   184→  // toast that reverts each email to its prior state.
   185→  function bulkUpdateWithUndo(
   186→    payload: Record<string, unknown>,
   187→    msg: string,
   188→    revertKey: "folder" | "snoozedUntil"
   189→  ) {
   190→    if (selected.size === 0) return;
   191→    const prev = new Map<string, string | null>();
   192→    for (const e of emails) {
   193→      if (selected.has(e.id)) {
   194→        prev.set(e.id, revertKey === "folder" ? e.folder : e.snoozedUntil);
   195→      }
   196→    }
   197→    fetch("/api/emails", {
   198→      method: "PATCH",
   199→      headers: { "Content-Type": "application/json" },
   200→      body: JSON.stringify({ ids: Array.from(selected), ...payload }),
   201→    })
   202→      .then((res) => {
   203→        if (!res.ok) throw new Error("Update failed");
   204→        setSelected(new Set());
   205→        invalidate();
   206→        showUndoToast(msg, () => revertBulk(revertKey, prev));
   207→      })
   208→      .catch(() => toast({ title: "Action failed", variant: "destructive" }));
   209→  }
   210→
   211→  async function revertBulk(
   212→    key: "folder" | "snoozedUntil",
   213→    prev: Map<string, string | null>
   214→  ) {
   215→    try {
   216→      await Promise.all(
   217→        Array.from(prev.entries()).map(([id, val]) =>
   218→          fetch(`/api/emails/${id}`, {
   219→            method: "PATCH",
   220→            headers: { "Content-Type": "application/json" },
   221→            body: JSON.stringify(
   222→              key === "folder"
   223→                ? { folder: val }
   224→                : { snoozedUntil: val }
   225→            ),
   226→          })
   227→        )
   228→      );
   229→      invalidate();
   230→      toast({ title: "Undo: restored", duration: 1500 });
   231→    } catch {
   232→      toast({ title: "Undo failed", variant: "destructive" });
   233→    }
   234→  }
   235→
   236→  async function bulkToggleLabel(label: string, checked: boolean) {
   237→    if (selected.size === 0) return;
   238→    const targets = emails.filter((e) => selected.has(e.id));
   239→    try {
   240→      await Promise.all(
   241→        targets.map((e) => {
   242→          const current = e.labels
   243→            .split(",")
   244→            .map((s) => s.trim())
   245→            .filter(Boolean);
   246→          const next = checked
   247→            ? Array.from(new Set([...current, label]))
   248→            : current.filter((l) => l !== label);
   249→          return fetch(`/api/emails/${e.id}`, {
   250→            method: "PATCH",
   251→            headers: { "Content-Type": "application/json" },
   252→            body: JSON.stringify({ labels: next.join(",") }),
   253→          });
   254→        })
   255→      );
   256→      invalidate();
   257→      toast({
   258→        title: checked ? `Added label "${label}"` : `Removed label "${label}"`,
   259→        duration: 1500,
   260→      });
   261→    } catch {
   262→      toast({ title: "Label update failed", variant: "destructive" });
   263→    }
   264→  }
   265→
   266→  // Labels active on ALL selected emails (for the bulk LabelMenu checkbox state)
   267→  const bulkActiveLabels = (() => {
   268→    const targets = emails.filter((e) => selected.has(e.id));
   269→    if (targets.length === 0) return new Set<string>();
   270→    return targets[0].labels
   271→      .split(",")
   272→      .map((s) => s.trim())
   273→      .filter(Boolean)
   274→      .filter((label) => targets.every((t) => t.labels.includes(label)))
   275→      .reduce<Set<string>>((set, l) => set.add(l), new Set());
   276→  })();
   277→
   278→  async function markReadAll(isRead: boolean) {
   279→    await bulkUpdate({ isRead }, `Marked ${isRead ? "read" : "unread"}`);
   280→  }
   281→
   282→  // Single-email row actions (with undo for archive/delete/snooze)
   283→  function singlePatch(id: string, payload: Record<string, unknown>) {
   284→    return fetch(`/api/emails/${id}`, {
   285→      method: "PATCH",
   286→      headers: { "Content-Type": "application/json" },
   287→      body: JSON.stringify(payload),
   288→    }).then((res) => {
   289→      if (!res.ok) throw new Error("Failed");
   290→      invalidate();
   291→    });
   292→  }
   293→
   294→  function rowArchive(email: Email) {
   295→    const prev = email.folder;
   296→    singlePatch(email.id, { folder: "ARCHIVE" })
   297→      .then(() => showUndoToast("Archived", () => singlePatch(email.id, { folder: prev })))
   298→      .catch(() => toast({ title: "Action failed", variant: "destructive" }));
   299→  }
   300→  function rowDelete(email: Email) {
   301→    const prev = email.folder;
   302→    singlePatch(email.id, { folder: "TRASH" })
   303→      .then(() => showUndoToast("Moved to Trash", () => singlePatch(email.id, { folder: prev })))
   304→      .catch(() => toast({ title: "Action failed", variant: "destructive" }));
   305→  }
   306→  function rowToggleRead(email: Email) {
   307→    singlePatch(email.id, { isRead: !email.isRead }).catch(() =>
   308→      toast({ title: "Action failed", variant: "destructive" })
   309→    );
   310→  }
   311→  function rowSnooze(email: Email, iso: string) {
   312→    singlePatch(email.id, { snoozedUntil: iso })
   313→      .then(() => showUndoToast("Snoozed", () => singlePatch(email.id, { snoozedUntil: null })))
   314→      .catch(() => toast({ title: "Action failed", variant: "destructive" }));
   315→  }
   316→  function rowUnsnooze(email: Email) {
   317→    singlePatch(email.id, { snoozedUntil: null }).catch(() =>
   318→      toast({ title: "Action failed", variant: "destructive" })
   319→    );
   320→  }
   321→
   322→  const title = searchQuery
   323→    ? `Results for "${searchQuery}"`
   324→    : selectedLabel
   325→    ? `Label: ${selectedLabel}`
   326→    : meta.label;
   327→
   328→  // Communication OS dedicated views (not flat email lists)
   329→  if (folder === "COMMAND_CENTER") {
   330→    return <CommandCenterView />;
   331→  }
   332→  if (folder === "COMMITMENTS") {
   333→    return <CommitmentsView />;
   334→  }
   335→  if (folder === "PEOPLE") {
   336→    return <PeopleView />;
   337→  }
   338→  if (folder === "ANALYTICS") {
   339→    return <AnalyticsView />;
   340→  }
   341→
   342→  return (
   343→    <div className="flex h-full flex-col bg-background">
   344→      {/* List header / toolbar */}
   345→      <div className="flex h-12 items-center gap-2 border-b border-border px-3 sm:px-4">
   346→        {selected.size > 0 ? (
   347→          <BulkToolbar
   348→            count={selected.size}
   349→            folder={folder}
   350→            onClear={() => setSelected(new Set())}
   351→            onArchive={() =>
   352→              bulkUpdateWithUndo({ folder: "ARCHIVE" }, "Archived", "folder")
   353→            }
   354→            onDelete={() => {
   355→              if (folder === "TRASH") {
   356→                // permanently delete selected
   357→                Promise.all(
   358→                  Array.from(selected).map((id) =>
   359→                    fetch(`/api/emails/${id}`, { method: "DELETE" })
   360→                  )
   361→                )
   362→                  .then(() => {
   363→                    setSelected(new Set());
   364→                    invalidate();
   365→                    toast({ title: "Deleted permanently" });
   366→                  })
   367→                  .catch(() =>
   368→                    toast({
   369→                      title: "Delete failed",
   370→                      variant: "destructive",
   371→                    })
   372→                  );
   373→              } else {
   374→                bulkUpdateWithUndo({ folder: "TRASH" }, "Moved to Trash", "folder");
   375→              }
   376→            }}
   377→            onMarkRead={() => markReadAll(true)}
   378→            onMarkUnread={() => markReadAll(false)}
   379→            onSnooze={(iso) =>
   380→              bulkUpdateWithUndo({ snoozedUntil: iso }, "Snoozed", "snoozedUntil")
   381→            }
   382→            onUnsnooze={() =>
   383→              bulkUpdateWithUndo({ snoozedUntil: null }, "Unsnoozed", "snoozedUntil")
   384→            }
   385→            activeLabels={bulkActiveLabels}
   386→            onToggleLabel={bulkToggleLabel}
   387→          />
   388→        ) : (
   389→          <div className="flex w-full items-center gap-2">
   390→            <Checkbox
   391→              checked={allChecked ? true : someChecked ? "indeterminate" : false}
   392→              onCheckedChange={toggleSelectAll}
   393→              aria-label="Select all"
   394→              className="ml-1"
   395→            />
   396→            <meta.icon className="h-4 w-4 text-muted-foreground" />
   397→            <h2 className="text-sm font-medium text-foreground">{title}</h2>
   398→            <span className="text-xs text-muted-foreground">
   399→              {emails.length > 0 && `${emails.length} ${emails.length === 1 ? "message" : "messages"}`}
   400→            </span>
   401→            {(folder === "TRASH" || folder === "SPAM") && emails.length > 0 && (
   402→              <Button
   403→                variant="ghost"
   404→                size="sm"
   405→                className="ml-2 h-8 gap-1 text-xs text-muted-foreground hover:text-destructive"
   406→                onClick={() => setEmptyConfirmOpen(true)}
   407→              >
   408→                <Trash2 className="h-3.5 w-3.5" />
   409→                Empty {folder === "TRASH" ? "trash" : "spam"} now
   410→              </Button>
   411→            )}
   412→            <div className="ml-auto flex items-center gap-1">
   413→              <TooltipProvider delayDuration={400}>
   414→                <Tooltip>
   415→                  <TooltipTrigger asChild>
   416→                    <Button
   417→                      variant="ghost"
   418→                      size="icon"
   419→                      className="h-8 w-8"
   420→                      onClick={() => refetch()}
   421→                      aria-label="Refresh"
   422→                    >
   423→                      <RefreshCw
   424→                        className={cn(
   425→                          "h-4 w-4",
   426→                          isFetching && "animate-spin"
   427→                        )}
   428→                      />
   429→                    </Button>
   430→                  </TooltipTrigger>
   431→                  <TooltipContent>Refresh</TooltipContent>
   432→                </Tooltip>
   433→              </TooltipProvider>
   434→              <Button variant="ghost" size="icon" className="h-8 w-8" aria-label="More">
   435→                <MoreVertical className="h-4 w-4" />
   436→              </Button>
   437→            </div>
   438→          </div>
   439→        )}
   440→      </div>
   441→
   442→      {/* Inbox category tabs */}
   443→      {showTabs && (
   444→        <div className="flex flex-shrink-0 items-center gap-1 overflow-x-auto border-b border-border bg-background px-2">
   445→          <InboxTabButton
   446→            active={inboxTab === "ALL"}
   447→            label="All"
   448→            count={tabCounts.ALL}
   449→            onClick={() => setInboxTab("ALL")}
   450→          />
   451→          <InboxTabButton
   452→            active={inboxTab === "PRIMARY"}
   453→            label="Primary"
   454→            count={tabCounts.PRIMARY}
   455→            dot="bg-primary"
   456→            onClick={() => setInboxTab("PRIMARY")}
   457→          />
   458→          <InboxTabButton
   459→            active={inboxTab === "PROMOTIONS"}
   460→            label="Promotions"
   461→            count={tabCounts.PROMOTIONS}
   462→            dot="bg-amber-500"
   463→            onClick={() => setInboxTab("PROMOTIONS")}
   464→          />
   465→          <InboxTabButton
   466→            active={inboxTab === "SOCIAL"}
   467→            label="Social"
   468→            count={tabCounts.SOCIAL}
   469→            dot="bg-purple-500"
   470→            onClick={() => setInboxTab("SOCIAL")}
   471→          />
   472→          <InboxTabButton
   473→            active={inboxTab === "UPDATES"}
   474→            label="Updates"
   475→            count={tabCounts.UPDATES}
   476→            dot="bg-emerald-500"
   477→            onClick={() => setInboxTab("UPDATES")}
   478→          />
   479→        </div>
   480→      )}
   481→
   482→      {/* Email rows */}
   483→      <div className="flex-1 overflow-y-auto">
   484→        {isError ? (
   485→          <div className="flex h-full items-center justify-center p-8 text-center text-sm text-muted-foreground">
   486→            Could not load emails.{" "}
   487→            <button
   488→              onClick={() => refetch()}
   489→              className="ml-1 text-primary underline"
   490→            >
   491→              Retry
   492→            </button>
   493→          </div>
   494→        ) : isLoading ? (
   495→          <EmailListSkeleton />
   496→        ) : emails.length === 0 ? (
   497→          <EmptyState folder={folder} search={!!searchQuery} />
   498→        ) : (
   499→          <DateGroupedEmailList
   500→            emails={emails}
   501→            selected={selected}
   502→            activeId={selectedEmailId}
   503→            onToggleSelect={toggleSelect}
   504→            onOpen={(id) => {
   505→              const target = emails.find((e) => e.id === id);
   506→              if (target && target.folder === "DRAFTS") {
   507→                openEditDraft(id);
   508→              } else {
   509→                setSelectedEmailId(id);
   510→                onOpenEmail(id);
   511→              }
   512→            }}
   513→            onArchive={rowArchive}
   514→            onDelete={rowDelete}
   515→            onToggleRead={rowToggleRead}
   516→            onSnooze={rowSnooze}
   517→            onUnsnooze={rowUnsnooze}
   518→          />
   519→        )}
   520→      </div>
   521→
   522→      <AlertDialog open={emptyConfirmOpen} onOpenChange={setEmptyConfirmOpen}>
   523→        <AlertDialogContent className="sm:max-w-[420px]">
   524→          <AlertDialogHeader>
   525→            <AlertDialogTitle>
   526→              Empty {folder === "TRASH" ? "trash" : "spam"}?
   527→            </AlertDialogTitle>
   528→            <AlertDialogDescription>
   529→              This will permanently delete all {emails.length}{" "}
   530→              {emails.length === 1 ? "message" : "messages"} in{" "}
   531→              {folder === "TRASH" ? "your trash" : "your spam"}. This action
   532→              cannot be undone.
   533→            </AlertDialogDescription>
   534→          </AlertDialogHeader>
   535→          <AlertDialogFooter>
   536→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   537→            <AlertDialogAction
   538→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   539→              onClick={handleEmptyFolder}
   540→            >
   541→              Delete forever
   542→            </AlertDialogAction>
   543→          </AlertDialogFooter>
   544→        </AlertDialogContent>
   545→      </AlertDialog>
   546→    </div>
   547→  );
   548→}

   550→const BUCKET_ORDER = ["Today", "Yesterday", "This week", "This month", "Earlier"] as const;
   551→
   552→function InboxTabButton({
   553→  active,
   554→  label,
   555→  count,
   556→  dot,
   557→  onClick,
   558→}: {
   559→  active: boolean;
   560→  label: string;
   561→  count: number;
   562→  dot?: string;
   563→  onClick: () => void;
   564→}) {
   565→  return (
   566→    <button
   567→      type="button"
   568→      onClick={onClick}
   569→      className={cn(

   585→function DateGroupedEmailList({
   586→  emails,
   587→  selected,
   588→  activeId,
   589→  onToggleSelect,
   590→  onOpen,
   591→  onArchive,
   592→  onDelete,
   593→  onToggleRead,
   594→  onSnooze,
   595→  onUnsnooze,
   596→}: {
   597→  emails: Email[];
   598→  selected: Set<string>;
   599→  activeId: string | null;
   600→  onToggleSelect: (id: string) => void;
   601→  onOpen: (id: string) => void;
   602→  onArchive: (email: Email) => void;
   603→  onDelete: (email: Email) => void;
   604→  onToggleRead: (email: Email) => void;
   605→  onSnooze: (email: Email, iso: string) => void;
   606→  onUnsnooze: (email: Email) => void;
   607→}) {
   608→  // Bucket emails preserving the date-desc order from the API
   609→  const groups = new Map<string, Email[]>();
   610→  for (const e of emails) {
   611→    const bucket = dateBucket(e.date);
   612→    if (!groups.has(bucket)) groups.set(bucket, []);
   613→    groups.get(bucket)!.push(e);
   614→  }
   615→
   616→  return (
   617→    <div>
   618→      {BUCKET_ORDER.filter((b) => groups.has(b)).map((bucket) => (
   619→        <section key={bucket}>
   620→          <div className="sticky top-0 z-10 flex h-7 items-center border-b border-border/60 bg-muted/40 px-4 backdrop-blur-sm">
   621→            <span className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   622→              {bucket}
   623→            </span>
   624→            <span className="ml-2 text-[10px] text-muted-foreground/70">
   625→              {groups.get(bucket)!.length}
   626→            </span>
   627→          </div>
   628→          <ul className="divide-y divide-border/60">
   629→            {groups.get(bucket)!.map((email) => (
   630→              <EmailRow
   631→                key={email.id}
   632→                email={email}
   633→                selected={selected.has(email.id)}
   634→                active={activeId === email.id}
   635→                onSelect={() => onToggleSelect(email.id)}
   636→                onOpen={() => onOpen(email.id)}
   637→                onArchive={() => onArchive(email)}
   638→                onDelete={() => onDelete(email)}
   639→                onToggleRead={() => onToggleRead(email)}
   640→                onSnooze={(iso) => onSnooze(email, iso)}
   641→                onUnsnooze={() => onUnsnooze(email)}
   642→              />
   643→            ))}
   644→          </ul>
   645→        </section>
   646→      ))}
   647→    </div>
   648→  );
   649→}
