   100→   100→  const [emptyConfirmOpen, setEmptyConfirmOpen] = useState(false);

   102→   102→  async function handleEmptyFolder() {

   104→   104→      await Promise.all(
   105→   105→        emails.map((e) =>
   106→   106→          fetch(`/api/emails/${e.id}`, { method: "DELETE" })

   152→   152→  function toggleSelectAll() {

   157→   157→  function toggleSelect(id: string) {

   159→   159→      const next = new Set(prev);
   160→   160→      if (next.has(id)) next.delete(id);
   161→   161→      else next.add(id);
   162→   162→      return next;
   163→   163→    });
   164→   164→  }
   165→   165→
   166→   166→  async function bulkUpdate(payload: Record<string, unknown>, msg: string) {
   167→   167→    if (selected.size === 0) return;
   168→   168→    try {
   169→   169→      const res = await fetch("/api/emails", {
   170→   170→        method: "PATCH",
   171→   171→        headers: { "Content-Type": "application/json" },
   172→   172→        body: JSON.stringify({ ids: Array.from(selected), ...payload }),
   173→   173→      });
   174→   174→      if (!res.ok) throw new Error("Update failed");
   175→   175→      setSelected(new Set());
   176→   176→      invalidate();
   177→   177→      toast({ title: msg });
   178→   178→    } catch {
   179→   179→      toast({ title: "Action failed", variant: "destructive" });
   180→   180→    }
   181→   181→  }
   182→   182→
   183→   183→  // Capture prev state per selected email, apply a bulk change, show an undo
   184→   184→  // toast that reverts each email to its prior state.
   185→   185→  function bulkUpdateWithUndo(
   186→   186→    payload: Record<string, unknown>,
   187→   187→    msg: string,
   188→   188→    revertKey: "folder" | "snoozedUntil"
   189→   189→  ) {
   190→   190→    if (selected.size === 0) return;
   191→   191→    const prev = new Map<string, string | null>();
   192→   192→    for (const e of emails) {
   193→   193→      if (selected.has(e.id)) {
   194→   194→        prev.set(e.id, revertKey === "folder" ? e.folder : e.snoozedUntil);
   195→   195→      }
   196→   196→    }
   197→   197→    fetch("/api/emails", {
   198→   198→      method: "PATCH",
   199→   199→      headers: { "Content-Type": "application/json" },
   200→   200→      body: JSON.stringify({ ids: Array.from(selected), ...payload }),
   201→   201→    })
   202→   202→      .then((res) => {
   203→   203→        if (!res.ok) throw new Error("Update failed");
   204→   204→        setSelected(new Set());
   205→   205→        invalidate();
   206→   206→        showUndoToast(msg, () => revertBulk(revertKey, prev));
   207→   207→      })
   208→   208→      .catch(() => toast({ title: "Action failed", variant: "destructive" }));
   209→   209→  }
   210→   210→
   211→   211→  async function revertBulk(
   212→   212→    key: "folder" | "snoozedUntil",
   213→   213→    prev: Map<string, string | null>
   214→   214→  ) {
   215→   215→    try {
   216→   216→      await Promise.all(
   217→   217→        Array.from(prev.entries()).map(([id, val]) =>
   218→   218→          fetch(`/api/emails/${id}`, {
   219→   219→            method: "PATCH",
   220→   220→            headers: { "Content-Type": "application/json" },
   221→   221→            body: JSON.stringify(
   222→   222→              key === "folder"
   223→   223→                ? { folder: val }
   224→   224→                : { snoozedUntil: val }
   225→   225→            ),
   226→   226→          })
   227→   227→        )
   228→   228→      );
   229→   229→      invalidate();
   230→   230→      toast({ title: "Undo: restored", duration: 1500 });
   231→   231→    } catch {
   232→   232→      toast({ title: "Undo failed", variant: "destructive" });
   233→   233→    }
   234→   234→  }
   235→   235→
   236→   236→  async function bulkToggleLabel(label: string, checked: boolean) {
   237→   237→    if (selected.size === 0) return;
   238→   238→    const targets = emails.filter((e) => selected.has(e.id));
   239→   239→    try {
   240→   240→      await Promise.all(
   241→   241→        targets.map((e) => {
   242→   242→          const current = e.labels
   243→   243→            .split(",")
   244→   244→            .map((s) => s.trim())
   245→   245→            .filter(Boolean);
   246→   246→          const next = checked
   247→   247→            ? Array.from(new Set([...current, label]))
   248→   248→            : current.filter((l) => l !== label);
   249→   249→          return fetch(`/api/emails/${e.id}`, {
   250→   250→            method: "PATCH",
   251→   251→            headers: { "Content-Type": "application/json" },
   252→   252→            body: JSON.stringify({ labels: next.join(",") }),
   253→   253→          });
   254→   254→        })
   255→   255→      );
   256→   256→      invalidate();
   257→   257→      toast({
   258→   258→        title: checked ? `Added label "${label}"` : `Removed label "${label}"`,
   259→   259→        duration: 1500,
   260→   260→      });
   261→   261→    } catch {
   262→   262→      toast({ title: "Label update failed", variant: "destructive" });
   263→   263→    }
   264→   264→  }
   265→   265→
   266→   266→  // Labels active on ALL selected emails (for the bulk LabelMenu checkbox state)
   267→   267→  const bulkActiveLabels = (() => {
   268→   268→    const targets = emails.filter((e) => selected.has(e.id));
   269→   269→    if (targets.length === 0) return new Set<string>();
   270→   270→    return targets[0].labels
   271→   271→      .split(",")
   272→   272→      .map((s) => s.trim())
   273→   273→      .filter(Boolean)
   274→   274→      .filter((label) => targets.every((t) => t.labels.includes(label)))
   275→   275→      .reduce<Set<string>>((set, l) => set.add(l), new Set());
   276→   276→  })();

   278→   278→  async function markReadAll(isRead: boolean) {

   280→   280→  }
   281→   281→
   282→   282→  // Single-email row actions (with undo for archive/delete/snooze)
   283→   283→  function singlePatch(id: string, payload: Record<string, unknown>) {
   284→   284→    return fetch(`/api/emails/${id}`, {
   285→   285→      method: "PATCH",
   286→   286→      headers: { "Content-Type": "application/json" },
   287→   287→      body: JSON.stringify(payload),
   288→   288→    }).then((res) => {
   289→   289→      if (!res.ok) throw new Error("Failed");
   290→   290→      invalidate();
   291→   291→    });
   292→   292→  }
   293→   293→
   294→   294→  function rowArchive(email: Email) {
   295→   295→    const prev = email.folder;
   296→   296→    singlePatch(email.id, { folder: "ARCHIVE" })
   297→   297→      .then(() => showUndoToast("Archived", () => singlePatch(email.id, { folder: prev })))
   298→   298→      .catch(() => toast({ title: "Action failed", variant: "destructive" }));
   299→   299→  }

   302→   302→    singlePatch(email.id, { folder: "TRASH" })
   303→   303→      .then(() => showUndoToast("Moved to Trash", () => singlePatch(email.id, { folder: prev })))
   304→   304→      .catch(() => toast({ title: "Action failed", variant: "destructive" }));
   305→   305→  }
   306→   306→  function rowToggleRead(email: Email) {
   307→   307→    singlePatch(email.id, { isRead: !email.isRead }).catch(() =>
   308→   308→      toast({ title: "Action failed", variant: "destructive" })
   309→   309→    );
   310→   310→  }

   312→   312→    singlePatch(email.id, { snoozedUntil: iso })
   313→   313→      .then(() => showUndoToast("Snoozed", () => singlePatch(email.id, { snoozedUntil: null })))
   314→   314→      .catch(() => toast({ title: "Action failed", variant: "destructive" }));
   315→   315→  }

   318→   318→      toast({ title: "Action failed", variant: "destructive" })
   319→   319→    );
   320→   320→  }
   321→   321→

   356→   356→                // permanently delete selected

   383→   383→              bulkUpdateWithUndo({ snoozedUntil: null }, "Unsnoozed", "snoozedUntil")
   384→   384→            }
   385→   385→            activeLabels={bulkActiveLabels}
   386→   386→            onToggleLabel={bulkToggleLabel}

   388→   388→        ) : (
   389→   389→          <div className="flex w-full items-center gap-2">

   391→   391→              checked={allChecked ? true : someChecked ? "indeterminate" : false}

   397→   397→            <h2 className="text-sm font-medium text-foreground">{title}</h2>
   398→   398→            <span className="text-xs text-muted-foreground">
   399→   399→              {emails.length > 0 && `${emails.length} ${emails.length === 1 ? "message" : "messages"}`}
   400→   400→            </span>
   401→   401→            {(folder === "TRASH" || folder === "SPAM") && emails.length > 0 && (
   402→   402→              <Button
   403→   403→                variant="ghost"
   404→   404→                size="sm"
   405→   405→                className="ml-2 h-8 gap-1 text-xs text-muted-foreground hover:text-destructive"
   406→   406→                onClick={() => setEmptyConfirmOpen(true)}
   407→   407→              >
   408→   408→                <Trash2 className="h-3.5 w-3.5" />
   409→   409→                Empty {folder === "TRASH" ? "trash" : "spam"} now
   410→   410→              </Button>
   411→   411→            )}
   412→   412→            <div className="ml-auto flex items-center gap-1">
   413→   413→              <TooltipProvider delayDuration={400}>
   414→   414→                <Tooltip>
   415→   415→                  <TooltipTrigger asChild>
   416→   416→                    <Button
   417→   417→                      variant="ghost"
   418→   418→                      size="icon"
   419→   419→                      className="h-8 w-8"
   420→   420→                      onClick={() => refetch()}
   421→   421→                      aria-label="Refresh"
   422→   422→                    >
   423→   423→                      <RefreshCw
   424→   424→                        className={cn(
   425→   425→                          "h-4 w-4",
   426→   426→                          isFetching && "animate-spin"
   427→   427→                        )}
   428→   428→                      />
   429→   429→                    </Button>
   430→   430→                  </TooltipTrigger>
   431→   431→                  <TooltipContent>Refresh</TooltipContent>
   432→   432→                </Tooltip>
   433→   433→              </TooltipProvider>
   434→   434→              <Button variant="ghost" size="icon" className="h-8 w-8" aria-label="More">
   435→   435→                <MoreVertical className="h-4 w-4" />
   436→   436→              </Button>
   437→   437→            </div>
   438→   438→          </div>
   439→   439→        )}
   440→   440→      </div>
   441→   441→
   442→   442→      {/* Inbox category tabs */}
   443→   443→      {showTabs && (
   444→   444→        <div className="flex flex-shrink-0 items-center gap-1 overflow-x-auto border-b border-border bg-background px-2">
   445→   445→          <InboxTabButton
   446→   446→            active={inboxTab === "ALL"}
   447→   447→            label="All"
   448→   448→            count={tabCounts.ALL}
   449→   449→            onClick={() => setInboxTab("ALL")}
   450→   450→          />
   451→   451→          <InboxTabButton
   452→   452→            active={inboxTab === "PRIMARY"}
   453→   453→            label="Primary"
   454→   454→            count={tabCounts.PRIMARY}
   455→   455→            dot="bg-primary"
   456→   456→            onClick={() => setInboxTab("PRIMARY")}
   457→   457→          />
   458→   458→          <InboxTabButton
   459→   459→            active={inboxTab === "PROMOTIONS"}
   460→   460→            label="Promotions"

   585→   585→function DateGroupedEmailList({
   586→   586→  emails,
   587→   587→  selected,
   588→   588→  activeId,
   589→   589→  onToggleSelect,
   590→   590→  onOpen,
   591→   591→  onArchive,
   592→   592→  onDelete,
   593→   593→  onToggleRead,
   594→   594→  onSnooze,
   595→   595→  onUnsnooze,
   596→   596→}: {
   597→   597→  emails: Email[];
   598→   598→  selected: Set<string>;
   599→   599→  activeId: string | null;
   600→   600→  onToggleSelect: (id: string) => void;
   601→   601→  onOpen: (id: string) => void;
   602→   602→  onArchive: (email: Email) => void;
   603→   603→  onDelete: (email: Email) => void;
   604→   604→  onToggleRead: (email: Email) => void;
   605→   605→  onSnooze: (email: Email, iso: string) => void;
   606→   606→  onUnsnooze: (email: Email) => void;
   607→   607→}) {
   608→   608→  // Bucket emails preserving the date-desc order from the API
   609→   609→  const groups = new Map<string, Email[]>();
   610→   610→  for (const e of emails) {
   611→   611→    const bucket = dateBucket(e.date);
   612→   612→    if (!groups.has(bucket)) groups.set(bucket, []);
   613→   613→    groups.get(bucket)!.push(e);
   614→   614→  }
   615→   615→
   616→   616→  return (
   617→   617→    <div>
   618→   618→      {BUCKET_ORDER.filter((b) => groups.has(b)).map((bucket) => (
   619→   619→        <section key={bucket}>
   620→   620→          <div className="sticky top-0 z-10 flex h-7 items-center border-b border-border/60 bg-muted/40 px-4 backdrop-blur-sm">
   621→   621→            <span className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground">
   622→   622→              {bucket}
   623→   623→            </span>
   624→   624→            <span className="ml-2 text-[10px] text-muted-foreground/70">
   625→   625→              {groups.get(bucket)!.length}
   626→   626→            </span>
   627→   627→          </div>
   628→   628→          <ul className="divide-y divide-border/60">
   629→   629→            {groups.get(bucket)!.map((email) => (
   630→   630→              <EmailRow
   631→   631→                key={email.id}
   632→   632→                email={email}
   633→   633→                selected={selected.has(email.id)}
   634→   634→                active={activeId === email.id}
   635→   635→                onSelect={() => onToggleSelect(email.id)}
   636→   636→                onOpen={() => onOpen(email.id)}
   637→   637→                onArchive={() => onArchive(email)}
   638→   638→                onDelete={() => onDelete(email)}
   639→   639→                onToggleRead={() => onToggleRead(email)}
   640→   640→                onSnooze={(iso) => onSnooze(email, iso)}
   641→   641→                onUnsnooze={() => onUnsnooze(email)}
   642→   642→              />
   643→   643→            ))}
   644→   644→          </ul>
   645→   645→        </section>
   646→   646→      ))}
   647→   647→    </div>
   648→   648→  );
   649→   649→}
